    stock_code VARCHAR(6) NOT NULL COMMENT '代码',
    trade_date VARCHAR(8) NOT NULL COMMENT '交易日期',

  `last_limitup_time` VARCHAR(10) COMMENT '最后封板时间',
  `breakout_count` TINYINT UNSIGNED COMMENT '炸板次数',

  KEY `idx_industry` (`industry`)
